/-
  RBMLTT.DependentCosts

  Skeleton for extending the RB-TT Lean development with
  dependent types (Π, Σ, Id, Nat, Vec) and compositional cost bounds.

  You will want to adapt:
  - `L` and its structure to the Cost lattice you already have in RB-TT;
  - `Ty` / `Ctx` / `Tm` to your existing syntax.
-/

namespace RBMLTT

-- Abstract cost lattice (you probably already have something like this).
structure CostLattice where
  (L     : Type)
  (le    : L → L → Prop)
  (bot   : L)
  (oplus : L → L → L)         -- monoidal sum
  (sup   : L → L → L)         -- binary join
  (le_refl  : ∀ a, le a a)
  (le_trans : ∀ {a b c}, le a b → le b c → le a c)
  (oplus_mono_left  : ∀ {a b c}, le a b → le (oplus a c) (oplus b c))
  (oplus_mono_right : ∀ {a b c}, le a b → le (oplus c a) (oplus c b))

infix:50 " ⊑ " => CostLattice.le
infixl:60 " ⊕ " => CostLattice.oplus

variable (CL : CostLattice)
abbrev L := CL.L

/-- Very simple dependent types (you will replace this with your real Ty). -/
inductive Ty : Type
  | nat   : Ty
  | vec   : Ty → Ty          -- `vec A` will be read as `Vec(A,n)` with `n` carried in the term
  | pi    : Ty → Ty → Ty     -- non-dependent Π for the skeleton
  | sigma : Ty → Ty → Ty
  | idty  : Ty → Ty          -- Id_A(a,b) collapsed for now

open Ty

/-- Context as a telescope of types (simplified). -/
inductive Ctx : Type
  | nil  : Ctx
  | snoc : Ctx → Ty → Ctx

open Ctx

/-- Variables as De Bruijn indices. -/
inductive Var : Ctx → Ty → Type
  | vz {Γ A}   : Var (snoc Γ A) A
  | vs {Γ A B} : Var Γ A → Var (snoc Γ B) A

/--
  Terms for the skeleton. You will want to enrich this with your
  real inductive family representation, universes, etc.
-/
inductive Tm : Ctx → Ty → Type
  | var   {Γ A}            : Var Γ A → Tm Γ A
  | lam   {Γ A B}          : Tm (snoc Γ A) B → Tm Γ (pi A B)
  | app   {Γ A B}          : Tm Γ (pi A B) → Tm Γ A → Tm Γ B
  | pair  {Γ A B}          : Tm Γ A → Tm Γ B → Tm Γ (sigma A B)
  | fst   {Γ A B}          : Tm Γ (sigma A B) → Tm Γ A
  | snd   {Γ A B}          : Tm Γ (sigma A B) → Tm Γ B
  | zero  {Γ}              : Tm Γ nat
  | succ  {Γ}              : Tm Γ nat → Tm Γ nat
  | natrec {Γ A}           :
      Tm Γ A →                                  -- z-case
      Tm Γ (pi nat (pi A A)) →                  -- s-case
      Tm Γ nat →
      Tm Γ A
  | nil   {Γ A}            : Tm Γ (vec A)
  | cons  {Γ A}            : Tm Γ A → Tm Γ (vec A) → Tm Γ (vec A)
  | vecrec {Γ A B}         :
      Tm Γ B →                                  -- nil-case
      Tm Γ (pi A (pi (vec A) (pi B B))) →       -- cons-case
      Tm Γ (vec A) →
      Tm Γ B
  | refl  {Γ A}            : Tm Γ A → Tm Γ (idty A)

/-
  Cost semantics: we assign to each term a *bound* in L, relative
  to an (implicit) interpretation of size information. For the
  skeleton we just take a simple structural definition.

  In your real development this should be:
  Γ ⊢ t : A with cost bound b  ↦  cost Γ A t ≤ b
-/
def costTm : {Γ : Ctx} → {A : Ty} → Tm CL Γ A → L CL
  | _, _, Tm.var _        => CL.bot
  | _, _, Tm.lam t        => costTm CL t          -- β-cost pushed to app
  | _, _, Tm.app f a      =>
      (costTm CL f) ⊕ (costTm CL a) ⊕ (/* δ_app */ CL.bot)
  | _, _, Tm.pair t u     =>
      (costTm CL t) ⊕ (costTm CL u) ⊕ (/* δ_pair */ CL.bot)
  | _, _, Tm.fst p        =>
      (costTm CL p) ⊕ (/* δ_fst */ CL.bot)
  | _, _, Tm.snd p        =>
      (costTm CL p) ⊕ (/* δ_snd */ CL.bot)
  | _, _, Tm.zero         => CL.bot
  | _, _, Tm.succ n       =>
      (costTm CL n) ⊕ (/* δ_succ */ CL.bot)
  | _, _, Tm.natrec z s n =>
      -- structurally recursive bound: O(n) in the size of `n`
      (costTm CL z) ⊕ (costTm CL s) ⊕ (costTm CL n) ⊕ (/* δ_natrec */ CL.bot)
  | _, _, Tm.nil          => CL.bot
  | _, _, Tm.cons x xs    =>
      (costTm CL x) ⊕ (costTm CL xs) ⊕ (/* δ_cons */ CL.bot)
  | _, _, Tm.vecrec z s v =>
      (costTm CL z) ⊕ (costTm CL s) ⊕ (costTm CL v) ⊕ (/* δ_vecrec */ CL.bot)
  | _, _, Tm.refl t       =>
      (costTm CL t) ⊕ (/* δ_refl */ CL.bot)

/-
  Example theorems you’d prove in the real repo:

  - cost_app: costTm (app f a) ⊑ costTm f ⊕ costTm a ⊕ δ_app
  - cost_natrec_linear: if n has size k, costTm (natrec z s n) is ≤ c·k + d
  - cost_vecrec_linear: same for vecrec over Vec(A,n).
-/

end RBMLTT
